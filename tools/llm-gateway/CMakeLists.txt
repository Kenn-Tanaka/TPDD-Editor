cmake_minimum_required(VERSION 3.20)
project(lightweight_llm_gateway VERSION 1.3.1 LANGUAGES CXX)

option(BUILD_CREDENTIAL_MANAGER "Build the wxWidgets credential manager" OFF)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(FetchContent)
FetchContent_Declare(cpp_httplib URL https://github.com/yhirose/cpp-httplib/archive/refs/tags/v0.18.7.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(nlohmann_json URL https://github.com/nlohmann/json/archive/refs/tags/v3.12.0.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(HTTPLIB_COMPILE OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(cpp_httplib nlohmann_json)

set(OPENSSL_USE_STATIC_LIBS TRUE)
set(ZLIB_USE_STATIC_LIBS TRUE)
find_package(OpenSSL REQUIRED)
find_package(ZLIB QUIET)
add_compile_definitions(CPPHTTPLIB_OPENSSL_SUPPORT)

add_library(gateway_core
  src/config.cpp src/credential_resolver.cpp src/gateway.cpp src/logger.cpp src/model_filter.cpp
  src/model_router.cpp src/provider.cpp src/proxy.cpp src/secret.cpp)
if(WIN32)
  target_sources(gateway_core PRIVATE src/credential/credential_windows.cpp)
  target_link_libraries(gateway_core PRIVATE Advapi32 Crypt32 Bcrypt Ws2_32)
elseif(UNIX)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LIBSECRET libsecret-1)
  if(LIBSECRET_FOUND)
    target_sources(gateway_core PRIVATE src/credential/credential_linux.cpp)
    target_include_directories(gateway_core PRIVATE ${LIBSECRET_INCLUDE_DIRS})
    target_link_directories(gateway_core PRIVATE ${LIBSECRET_LIBRARY_DIRS})
    target_link_libraries(gateway_core PRIVATE ${LIBSECRET_LIBRARIES})
    target_compile_options(gateway_core PRIVATE ${LIBSECRET_CFLAGS_OTHER})
  else()
    target_sources(gateway_core PRIVATE src/credential/credential_linux_unavailable.cpp)
  endif()
endif()
target_include_directories(gateway_core PUBLIC include)
target_link_libraries(gateway_core PUBLIC httplib::httplib nlohmann_json::nlohmann_json)
target_link_libraries(gateway_core PUBLIC OpenSSL::SSL OpenSSL::Crypto)
if(TARGET ZLIB::ZLIB)
  target_link_libraries(gateway_core PUBLIC ZLIB::ZLIB)
endif()
if(MSVC)
  target_compile_options(gateway_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(gateway_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(llm-gateway src/main.cpp)
target_link_libraries(llm-gateway PRIVATE gateway_core)
if(MINGW OR CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_link_options(llm-gateway PRIVATE -static -static-libgcc -static-libstdc++)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(sse-poc tests/sse_poc.cpp)
  target_link_libraries(sse-poc PRIVATE httplib::httplib)
  add_test(NAME sse_poc COMMAND sse-poc)
  set_tests_properties(sse_poc PROPERTIES TIMEOUT 15)

  add_executable(gateway-tests tests/gateway_tests.cpp tests/mock_provider.cpp)
  target_link_libraries(gateway-tests PRIVATE gateway_core)
  add_test(NAME gateway_tests COMMAND gateway-tests)
  set_tests_properties(gateway_tests PROPERTIES TIMEOUT 30)
  if(WIN32)
    add_executable(credential-interop-test tests/test_credential_interop_windows.cpp)
    target_link_libraries(credential-interop-test PRIVATE gateway_core Advapi32)
    add_test(NAME credential_interop_windows COMMAND credential-interop-test)
    add_executable(manager-credential-test tests/test_manager_credential_windows.cpp manager/src/credential_windows.cpp manager/src/validation.cpp)
    target_include_directories(manager-credential-test PRIVATE manager/include)
    target_link_libraries(manager-credential-test PRIVATE gateway_core Advapi32)
    add_test(NAME manager_gateway_credential_interop COMMAND manager-credential-test)
  endif()
endif()

install(TARGETS llm-gateway RUNTIME DESTINATION bin)
install(FILES config/gateway.example.json DESTINATION share/llm-gateway)

if(BUILD_CREDENTIAL_MANAGER)
  add_subdirectory(manager)
endif()
